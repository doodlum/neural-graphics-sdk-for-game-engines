# Loader shims that let NGX accept the NSS proxy core (see loader_shim.cpp). One DLL per exports/<name>.txt.
cmake_minimum_required(VERSION 3.20)
project(deckyupscaler_loader CXX)

file(GLOB SHIM_LISTS "${CMAKE_CURRENT_SOURCE_DIR}/exports/*.txt")
foreach(list ${SHIM_LISTS})
    get_filename_component(name "${list}" NAME_WE)
    file(STRINGS "${list}" exports)
    set(gen "${CMAKE_CURRENT_BINARY_DIR}/${name}")
    set(inc "#define SHIM_EXPORTS(X)")
    set(def "LIBRARY ${name}.dll\nEXPORTS\n")
    foreach(e ${exports})
        string(APPEND inc " X(${e})")
        string(APPEND def "    ${e}\n")
    endforeach()
    file(WRITE "${gen}/shim_exports.inc" "${inc}\n")
    file(WRITE "${gen}/${name}.def" "${def}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${list}")

    add_library(shim_${name} SHARED loader_shim.cpp "${gen}/${name}.def")
    set_target_properties(shim_${name} PROPERTIES PREFIX "" OUTPUT_NAME ${name} RUNTIME_OUTPUT_DIRECTORY "${gen}")
    target_include_directories(shim_${name} PRIVATE "${gen}")
    target_compile_features(shim_${name} PRIVATE cxx_std_17)
    target_compile_options(shim_${name} PRIVATE -O2 -Wall)
    target_compile_definitions(shim_${name} PRIVATE WIN32_LEAN_AND_MEAN UNICODE _UNICODE SHIM_TARGET=L"${name}.dll")
    target_link_options(shim_${name} PRIVATE -s)
endforeach()

# FFX forwarder (ffx_forwarder.cpp): stands in for a game's amd_fidelityfx_vk.dll and sends its FSR upscaling to the proxy.
add_library(ffxfwd_vk SHARED ffx_forwarder.cpp)
set_target_properties(ffxfwd_vk PROPERTIES PREFIX "" OUTPUT_NAME amd_fidelityfx_vk RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/ffx")
target_compile_features(ffxfwd_vk PRIVATE cxx_std_17)
target_compile_options(ffxfwd_vk PRIVATE -O2 -Wall)
target_compile_definitions(ffxfwd_vk PRIVATE WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_options(ffxfwd_vk PRIVATE -s -static-libgcc -static-libstdc++)
